// CONSTRUCTOR public Inmueble(String id, String dir, double pre, double area) { this.idReferencia = id; this.direccion = dir; this.precioBase = pre; this.areaMetros = area; this.estado = "Disponible"; this.fechaRegistro = LocalDate.now(); } // MÉTODOS DE LÓGICA AGREGADOS public double calcularComision() { return this.precioBase * 0.03; } public void aplicarDescuento(double porcentaje) { if (porcentaje > 0 && porcentaje <= 15) { this.precioBase -= (this.precioBase * (porcentaje / 100)); } } public void mostrarFicha() { System.out.println("Inmueble ID: " + idReferencia + " | Estado: " + estado); }
